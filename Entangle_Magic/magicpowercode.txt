def M2(psiout):
    traces=[]
    I=np.array([[1,0],[0,1]])
    X=np.array([[0,1],[1,0]])
    Y=np.array([[0,-1j],[1j,0]])
    Z=np.array([[1,0],[0,-1]])
    pauli=np.array([I,X,Y,Z])
    # print(pauli.shape)
    # print(pauli[3])
    
    rho=np.outer(psiout,psiout.conj())
    
    for i in range(4):
        for j in range(4):
            # print('i',i,' j',j)
            P=np.kron(pauli[i],pauli[j])
            # print('P shape',P.shape)
            trace=np.trace(rho@P)**4
            traces.append(trace)
    xi2=1/4*np.sum(traces)
    return -np.log(xi2)
